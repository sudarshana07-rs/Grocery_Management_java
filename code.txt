import java.util.Scanner;
import java.util.Random;
// Class: Main
public class demo{

     // Creating total variable 
     static int total=0;    
     static String name;  
     
    // Main method
    public static void main(String args[ ]){
        // Intro Statements
        Scanner sc= new Scanner(System.in);
        System.out.println("==== Grocery Mangement System ===");
        System.out.println();
        System.out.println("--- Welcome to XYZ Supermarket ---");
        System.out.println();

        // Customer Details
        System.out.println("Enter your name: ");
        name = sc.nextLine();
        System.out.println("------------------------------------");

        // To display the Grocery List
        items myItems = new items();
        System.out.println("Do you want to see the Grocery list (Yes/No): ");
        String see_items= sc.nextLine();
        if(see_items.equals("Yes")){
            myItems.displayItems();
        }
        else{
        System.out.println("Check the Grocery List");
        }      

        // While loop
        while(true){
            
        // Product input
        System.out.println("--------------------------------");
        System.out.println("Enter the Product number (or press 6 to Remove, 7 for Bill, 8 to Exit: ");
        int choice = sc.nextInt();
        
        // For choice 6: Remove Item
        if (choice==6){
            remove_item myremove= new remove_item();
            myremove.removeItems(sc);
            continue;
        }

        // For Choice 7: Display Bill                        
        if(choice==7){            
            billing myBilling = new billing();
            myBilling.displayBilling();
            continue;
        }            
        // For choice 8: Exit   
        if(choice==8){                                     
            System.out.println("Thank You for Shoping!!!");
            break;                                         
        }                                                  

        // Quantity input                          
        System.out.println("Enter the Quantity: ");
        int quantity = sc.nextInt();   

        // Switch Case
        switch (choice) {
                case 1:
                    total += 60*quantity;
                    System.out.println(quantity+ " kg of Rice added Successfully!!!");
                    break;
                case 2:                                                           
                    total += 50*quantity;                                           
                    System.out.println(quantity+ " kg of Sugar added Successfully!!!");
                    break;    
                case 3:                                                           
                    total += 150*quantity;                                           
                    System.out.println(quantity+ " kg of Dal added Successfully!!!");
                    break;            
                case 4:                                                                                                       
                    total += 50*quantity;                                           
                    System.out.println(quantity+ " kg of Wheat added Successfully!!!");
                    break; 
                case 5:                                                               
                    total += 60*quantity;                                               
                    System.out.println(quantity+ " litre of Milk added Successfully!!!");
                    break;              
                default:
                    System.out.println("INVALID PRODUCT !!!");
                } 
        }
    }
}

// Class: Billing
class billing{
    public void displayBilling(){

        // Random class object creation
        Random rand = new Random();
        // Gives random bill number from 1000 to 9999
        int billNumber = rand.nextInt(9000)+1000;

        System.out.println("============================");
        System.out.println("        XYZ SUPERMARKET       ");

}

// Class : REMOVE
class remove_item{
    void removeItems(Scanner sc){          
        int removeChoice;
        do{                           
            System.out.println();                                        
            System.out.println("---- REMOVE ITEM ---");               
            System.out.println("Sl.no\tProduct\tUnit\tPrice");       
            System.out.println("1\tRice\t1 kg\tRs.60");              
            System.out.println("2\tSugar\t1 kg\tRs.50");             
            System.out.println("3\tDal\t1 kg\tRs.150");              
            System.out.println("4\tWheat\t1 kg\tRs.45");             
            System.out.println("5\tMilk\t1 litre\tRs.60");           
            System.out.println();                                    

            System.out.println("Enter the product number to remove");
            removeChoice = sc.nextInt();       
            
        if(removeChoice == 0){
            break;
        }
                    
        if (removeChoice >= 1 && removeChoice<=5){
            // Removing Item Quantity
            System.out.println("Enter the quantity to remove: ");
            int removeQty = sc.nextInt();                        
        switch(removeChoice){
            case 1: 
                demo.total -= 60*removeQty;                                                         
                System.out.println(removeQty+ " kg of Rice removed Successfully!!!");
                break;
            case 2:                                                                 
                demo.total -= 50*removeQty;                                              
                System.out.println(removeQty+" kg of Sugar removed Successfully!!!");
                break;
            case 3:                                                                
                demo.total -= 150*removeQty;                                             
                System.out.println(removeQty+" kg of Dal removed Successfully!!!");
                break;           
            case 4:                                                                   
                demo.total -= 50*removeQty;                                                 
                System.out.println(removeQty+" kg of Wheat removed Successfully!!!");
                break;     
            case 5:                                                                
                demo.total -= 60*removeQty;                                                 
                System.out.println(removeQty+" litre of Milk removed Successfully!!!");
                break;

        }
    }
    else{
        System.out.println("INVALID ITEM TO REMOVE !!!");
    }
        } while (removeChoice != 0);
    }
}
